            <h2 id="modal-titulo">Novo Registo</h2>
            <select id="categoria"></select>
            <input type="number" id="valor" placeholder="Valor em Kwanza" inputmode="numeric">
            <button class="btn-entrada" id="btn-salvar" style="width: 100%;" onclick="salvarRegisto()">CONFIRMAR</button>
            <button onclick="fecharModal()" style="background:none; color: gray; width: 100%; margin-top: 10px;">Cancelar</button>
        </div>
    </div>

    <script>
        let dados = JSON.parse(localStorage.getItem('conforto_lazer_db')) || [];
        let tipoAtual = '';

        const categorias = {
            entrada: ['Almoço', 'Bar / Esplanada', 'Petiscos', 'Combos'],
            saida: ['Compras', 'Gás / Carvão', 'Energia', 'Água', 'Transporte', 'Outros']
        };

        function atualizarUI() {
            const hoje = new Date().toLocaleDateString('pt-PT');
            document.getElementById('data-hoje').innerText = hoje;

            const registrosHoje = dados.filter(d => d.data === hoje);
            const entradas = registrosHoje.filter(d => d.tipo === 'entrada').reduce((a, b) => a + b.valor, 0);
            const saidas = registrosHoje.filter(d => d.tipo === 'saida').reduce((a, b) => a + b.valor, 0);
            
            document.getElementById('total-entradas').innerText = entradas.toLocaleString() + ' Kz';
            document.getElementById('total-saidas').innerText = saidas.toLocaleString() + ' Kz';
            document.getElementById('saldo-dia').innerText = (entradas - saidas).toLocaleString() + ' Kz';

            const lista = document.getElementById('lista-atividades');
            lista.innerHTML = registrosHoje.slice().reverse().slice(0, 8).map(r => `
                <div class="item-lista">
                    <span>${r.hora.slice(0,5)} - ${r.cat}</span>
                    <b class="${r.tipo === 'entrada' ? 'positivo' : 'negativo'}">${r.valor} Kz</b>
                </div>
            `).join('');

            localStorage.setItem('conforto_lazer_db', JSON.stringify(dados));
            renderGrafico(entradas, saidas);
        }

        function abrirModal(tipo) {
            tipoAtual = tipo;
            const select = document.getElementById('categoria');
            select.innerHTML = categorias[tipo].map(c => `<option value="${c}">${c}</option>`).join('');
            document.getElementById('modal-titulo').innerText = tipo === 'entrada' ? 'Registar Ganho' : 'Registar Gasto';
            document.getElementById('btn-salvar').className = tipo === 'entrada' ? 'btn-entrada' : 'btn-saida';
            document.getElementById('modal').style.display = 'flex';
            document.getElementById('valor').focus();
        }

        function fecharModal() { document.getElementById('modal').style.display = 'none'; }

        function salvarRegisto() {
            const valorInput = document.getElementById('valor').value;
            if (!valorInput) return;

            dados.push({
                tipo: tipoAtual,
                cat: document.getElementById('categoria').value,
                valor: parseFloat(valorInput),
                data: new Date().toLocaleDateString('pt-PT'),
                hora: new Date().toLocaleTimeString('pt-PT')
            });

            fecharModal();
            document.getElementById('valor').value = '';
            atualizarUI();
        }

        function gerarRelatorioExcel() {
            if(dados.length === 0) return alert("Sem dados para exportar.");
            
            const worksheet = XLSX.utils.json_to_sheet(dados);
            const workbook = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(workbook, worksheet, "Financeiro");
            
            const dataAtual = new Date().toISOString().slice(0,10);
            XLSX.writeFile(workbook, `Conforto_Lazer_Relatorio_${dataAtual}.xlsx`);
        }

        let chart;
        function renderGrafico(e, s) {
            const ctx = document.getElementById('graficoDia').getContext('2d');
            if (chart) chart.destroy();
            chart = new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: ['Entradas', 'Saídas'],
                    datasets: [{ data: [e, s], backgroundColor: ['#27ae60', '#e74c3c'], borderWidth: 0 }]
                },
                options: { plugins: { legend: { position: 'bottom' } } }
            });
        }

        atualizarUI();
    </script>
